
    document.getElementById('stat-total').textContent     = stats.total;
    document.getElementById('stat-confirmed').textContent  = stats.confirmed;
    document.getElementById('stat-in_progress').textContent = stats.in_progress;
    document.getElementById('stat-rejected').textContent   = stats.rejected;

    // Drive count hint
    const driveCount = applications.filter(a => a.source === 'drive').length;
    const dc = document.getElementById('stat-drive-count');
    if (dc) dc.textContent = driveCount > 0 ? `${driveCount} from Drive` : '';

function renderFunnel(stats) {
    const el = document.getElementById('funnel-container');
    if (stats.total === 0) {
        el.innerHTML = '<div class="empty-state py-6"><p class="text-sm text-gray-400">No data yet</p></div>';
        return;
    }
    const steps = [
        { label:'Applied',     count: stats.total,       color:'bg-slate-400',  pct: 100 },
        { label:'Confirmed',   count: stats.confirmed,   color:'bg-blue-400',   pct: stats.total ? Math.round(stats.confirmed/stats.total*100)   : 0 },
        { label:'In Progress', count: stats.in_progress, color:'bg-amber-400',  pct: stats.total ? Math.round(stats.in_progress/stats.total*100) : 0 },
        { label:'Rejected',    count: stats.rejected,    color:'bg-red-400',    pct: stats.total ? Math.round(stats.rejected/stats.total*100)    : 0 },
    ];
    el.innerHTML = steps.map(s => `
        <div class="funnel-step">
            <span class="funnel-label">${s.label}</span>
            <div class="funnel-bar-wrap">
                <div class="progress-bar">
                    <div class="progress-fill ${s.color}" style="width:${s.pct}%"></div>
                </div>
            </div>
            <span class="funnel-count">${s.count}</span>
        </div>
    `).join('');
}

function renderTable() {
    const search = (document.getElementById('search-input')?.value || '').toLowerCase();
    const el = document.getElementById('app-table-body');

    let filtered = applications.filter(a => {
        // Normalise legacy statuses for filtering
        const normStatus = ['confirmed','in_progress','rejected'].includes(a.status)
            ? a.status
            : (a.status === 'rejected' ? 'rejected' : (a.status === 'confirmed' || a.status === 'applied' || a.status === 'offer' ? 'confirmed' : 'in_progress'));
        const matchFilter = activeFilter === 'all' || normStatus === activeFilter;
        const matchSearch = !search ||
            a.company.toLowerCase().includes(search) ||
            a.role.toLowerCase().includes(search) ||
            (a.notes||'').toLowerCase().includes(search);
        return matchFilter && matchSearch;
    });

    if (filtered.length === 0) {
        el.innerHTML = `<div class="empty-state"><svg class="w-12 h-12" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="1.5" d="M9.172 16.172a4 4 0 015.656 0M9 10h.01M15 10h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z"></path></svg><p class="font-semibold text-gray-400 mt-2">No results found</p></div>`;
        return;
    }

    const statusBadge = (s) => {
        const map = {
            confirmed:   'badge-applied Confirmed',
            in_progress: 'badge-interview In Progress',
            rejected:    'badge-rejected Rejected',
            // legacy statuses mapped to new ones
            applied:  'badge-applied Confirmed',
            interview:'badge-interview In Progress',
            offer:    'badge-offer Offer',
            pending:  'badge-pending Pending',
            ghosted:  'badge-ghosted Ghosted',

        const parts = (map[s]||'badge-pending Unknown').split(' ');
        return `<span class="status-badge ${parts[0]}">${parts.slice(1).join(' ')}</span>`;
    };

    el.innerHTML = filtered.map(a => `
        <div class="app-row cursor-pointer ${selectedAppId === a.id ? 'bg-indigo-50' : ''}" onclick="selectApp('${a.id}')">
            <div class="min-w-0">

                <span>${a.date || '—'}</span>
            </div>
            <div>${statusBadge(a.status)}</div>
            <div class="text-xs text-gray-600 truncate">${escHtml(a.nextStep || '—')}</div>
            <button type="button" onclick="event.stopPropagation(); deleteApp('${a.id}')" class="w-7 h-7 flex items-center justify-center text-gray-300 hover:text-red-500 hover:bg-red-50 rounded-lg transition-colors cursor-pointer" title="Delete">
                <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"></path></svg>
            </button>
        </div>
    `).join('');
}

async function fetchGeminiModels(silent = false) {
    const apiKey = document.getElementById('global-api-key-input')?.value?.trim();
    if (!apiKey) { showToast('Enter your Gemini API Key first.', true); return; }

    const btn = document.getElementById('refresh-models-btn');
    if (btn && !silent) btn.textContent = '↻ Loading...';

    try {
        const resp = await fetch(
            `https://generativelanguage.googleapis.com/v1beta/models?pageSize=50&key=${apiKey}`
        );
        const data = await resp.json();
        if (!resp.ok) throw new Error(data.error?.message || `HTTP ${resp.status}`);

        // Filter to text-generation capable models only, exclude audio/video/embedding/tts
        const exclude = ['embedding', 'tts', 'speech', 'audio', 'video', 'image', 'vision', 'aqa', 'retrieval'];
        const models = (data.models || [])
            .filter(m => {
                const name = m.name.toLowerCase();
                const desc = (m.description || '').toLowerCase();
                // Must support generateContent
                const supportsGenerate = (m.supportedGenerationMethods || []).includes('generateContent');
                // Exclude non-text models
                const isExcluded = exclude.some(e => name.includes(e) || desc.includes(e));
                return supportsGenerate && !isExcluded;
            })
            .map(m => ({
                id: m.name.replace('models/', ''),
                display: m.displayName || m.name.replace('models/', ''),
                description: m.description || ''
            }))
            .sort((a, b) => {
                // Sort: gemini-3 first, then 2.5, then others
                const rank = id => id.startsWith('gemini-3') ? 0 : id.startsWith('gemini-2.5') ? 1 : 2;
                return rank(a.id) - rank(b.id) || a.id.localeCompare(b.id);
            });

        if (!models.length) {
            showToast('No text models found for this API key.', true);
            return;
        }

        const sel = document.getElementById('global-ai-model-input');
        if (!sel) return;

        const currentVal = sel.value;

        // Rebuild options grouped by series
        sel.innerHTML = '';
        const groups = {
            'Gemini 3': models.filter(m => m.id.startsWith('gemini-3')),
            'Gemini 2.5': models.filter(m => m.id.startsWith('gemini-2.5')),
            'Other': models.filter(m => !m.id.startsWith('gemini-3') && !m.id.startsWith('gemini-2.5')),
        };

        for (const [groupName, groupModels] of Object.entries(groups)) {
            if (!groupModels.length) continue;
            const og = document.createElement('optgroup');
            og.label = `── ${groupName} ──`;
            groupModels.forEach(m => {
                const opt = document.createElement('option');
                opt.value = m.id;
                // Add helpful suffix hints
                const hint = m.id.includes('flash-lite') ? ' — budget' :
                             m.id.includes('flash') ? ' — fast' :
                             m.id.includes('pro') ? ' — powerful' : '';
                opt.textContent = m.id + hint;
                og.appendChild(opt);
            });
            sel.appendChild(og);
        }

        // Restore selection or default to gemini-2.5-flash
        const preferred = currentVal || 'gemini-2.5-flash';
        const exists = Array.from(sel.options).some(o => o.value === preferred);
        sel.value = exists ? preferred : sel.options[0]?.value;
        setLocalStorageSafe('gemini_model', sel.value);

        if (!silent) showToast(`Loaded ${models.length} models`);
    } catch(e) {
        showToast('Failed to load models: ' + e.message, true);
        console.error('[Models]', e);
    } finally {
        if (btn) btn.textContent = '↻ Refresh';
    }
}

function handleImageUpload(event) {
    try {
        const file = event.target.files[0];
        if (!file) return;
        saveState();
        const reader = new FileReader();
        reader.onload = e => {
            const img = new Image();
            img.onload = () => {
                const canvas = document.createElement('canvas');
                let [w,h] = [img.width, img.height];
                const MAX = 300;
                if (w > h) { if (w>MAX) { h*=MAX/w; w=MAX; } } else { if (h>MAX) { w*=MAX/h; h=MAX; } }
                canvas.width=w; canvas.height=h;
                canvas.getContext('2d').drawImage(img,0,0,w,h);
                const b64 = canvas.toDataURL('image/jpeg',0.8);
                const i1 = document.getElementById('profile-img');
                const i2 = document.getElementById('tc-profileImage');
                if(i1) i1.src=b64; if(i2) i2.src=b64;
            };
            img.src = e.target.result;
        };
        reader.readAsDataURL(file);
        event.target.value='';
    } catch(e) { showToast('Could not process image.',true); }
}

function importHTML(event) {
    try {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = e => {
            try {
                const doc = new DOMParser().parseFromString(e.target.result,'text/html');
                const newResume = doc.getElementById('resume-main-container') || doc.querySelector('main');
                if (newResume) {
                    saveState();
                    document.getElementById('resume-main-container').innerHTML = newResume.innerHTML;
                    const nc = doc.getElementById('talent-card-view');
                    const cc = document.getElementById('talent-card-view');
                    if (nc && cc) cc.innerHTML = nc.innerHTML;
                    showToast('Resume loaded!');
                    if (typeof lucide !== 'undefined') lucide.createIcons();
                } else { showToast('Invalid file.',true); }
            } catch(err) { showToast('Failed to process HTML.',true); }
            finally { event.target.value=''; }
        };
        reader.readAsText(file);
    } catch(e) { showToast('Failed to load file.',true); }
}

async function downloadHTML() {
    try {
        const blob = new Blob(["<!DOCTYPE html>\n" + document.documentElement.outerHTML], {type:'text/html'});
        if (window.showSaveFilePicker) {
            const h = await window.showSaveFilePicker({suggestedName:'Resume_Interactive.html',types:[{description:'HTML File',accept:{'text/html':['.html']}}]});
            const w = await h.createWritable(); await w.write(blob); await w.close();
            showToast('HTML saved!');
        } else {
            const url = URL.createObjectURL(blob);
            const a = Object.assign(document.createElement('a'),{href:url,download:'Resume_Interactive.html'});
            document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url);
            showToast('HTML downloaded!');
        }
    } catch(e) { if(e.name!=='AbortError') showToast('Failed to save.',true); }
}

function dragStart(e) { draggedSkill=e.currentTarget; setTimeout(()=>e.currentTarget.classList.add('opacity-50'),0); }
function dragEnd(e) { e.currentTarget.classList.remove('opacity-50'); draggedSkill=null; saveState(); }
function dragOver(e) {
    e.preventDefault();
    const t = e.target.closest('span[draggable="true"]');
    if (t && t!==draggedSkill && draggedSkill?.parentElement===t.parentElement) {
        const r = t.getBoundingClientRect();
        if (e.clientX > r.left+r.width/2) t.after(draggedSkill); else t.before(draggedSkill);
    }
}

async function fetchWithRetry(url, options, maxRetries=5) {
    let delay=1000;
    for (let i=0; i<maxRetries; i++) {
        try {
            const r = await fetch(url,options);
            const d = await r.json();
            if (!r.ok) throw new Error(d.error?.message||`HTTP ${r.status}`);
            return d;
        } catch(err) {
            if(i===maxRetries-1) throw err;
            await new Promise(res=>setTimeout(res,delay)); delay*=2;
        }
    }
}

        if(sc&&aiData.skills){const existing=[...sc.querySelectorAll('span.outline-none')].map(s=>s.innerText.trim().toLowerCase());aiData.skills.slice(0,10).forEach(skill=>{if(!existing.includes(skill.toLowerCase())){const el=document.createElement('span');el.className='suggested-item bg-purple-50 border border-purple-300 text-purple-800 px-3 py-1 rounded-md text-sm font-medium relative group/suggestion flex items-center shadow-sm';el.innerHTML=`<span contenteditable="true" class="outline-none min-w-[20px] cursor-text" onmousedown="event.stopPropagation()">${skill}</span><div class="absolute -top-2.5 -right-4 flex gap-0.5 opacity-0 group-hover/suggestion:opacity-100 no-print z-20"><button type="button" onclick="approveSkill(this)" class="bg-green-500 hover:bg-green-600 text-white rounded-full w-5 h-5 flex items-center justify-center text-xs shadow font-bold cursor-pointer">✓</button><button type="button" onclick="removeElement(this.closest('.suggested-item'))" class="bg-red-500 hover:bg-red-600 text-white rounded-full w-5 h-5 flex items-center justify-center text-xs shadow font-bold cursor-pointer">×</button></div>`;sc.appendChild(el);}});}
        if(aiData.experiences){document.querySelectorAll('#resume-main-container #experience-container > div').forEach(block=>{const cn=block.querySelector('h3 span:nth-child(1)');const tn=block.querySelector('h4');if(cn&&tn){const match=aiData.experiences.find(e=>e.company===cn.innerText.trim()&&e.title===tn.innerText.trim());if(match?.bullets){const ul=block.querySelector('ul');const helper=ul.querySelector('.helper-bullet');match.bullets.forEach(bt=>{const li=document.createElement('li');li.className='suggested-item relative group/suggestion pr-8 mb-2';li.innerHTML=`<span contenteditable="true" class="block outline-none text-purple-900 bg-purple-50 px-2 py-1.5 rounded border border-purple-200">${bt}</span><div class="absolute top-1 right-0 flex gap-0.5 opacity-0 group-hover/suggestion:opacity-100 no-print z-20"><button type="button" onclick="approveBullet(this)" class="bg-green-500 hover:bg-green-600 text-white rounded w-5 h-5 flex items-center justify-center text-xs shadow cursor-pointer">✓</button><button type="button" onclick="removeElement(this.closest('.suggested-item'))" class="bg-red-500 hover:bg-red-600 text-white rounded w-5 h-5 flex items-center justify-center text-xs shadow cursor-pointer">×</button></div>`;if(helper)ul.insertBefore(li,helper);else ul.appendChild(li);});}}});}
        toggleView('resume');
        const ms=document.getElementById('global-match-score');
        if(ms){ms.innerText=`Match: ${aiData.match_score||0}%`;ms.classList.remove('hidden');}
        closeAiModal(); checkPendingSuggestions(); showToast('Suggestions added!');
    } catch(err) {
        document.getElementById('generate-ai-btn').classList.remove('hidden');
        document.getElementById('job-description-input').disabled=false;
        errMsg.innerText='Error: '+err.message; errMsg.classList.remove('hidden');
    } finally {
        document.getElementById('ai-loading').classList.add('hidden');
        document.getElementById('ai-loading').classList.remove('flex');
    }
}

function checkPendingSuggestions() {
    const has=document.querySelectorAll('.suggested-item').length>0;
    const btn=document.getElementById('approve-all-suggestions-btn');
    if(btn){has?(btn.classList.remove('hidden'),btn.classList.add('flex')):(btn.classList.add('hidden'),btn.classList.remove('flex'));}
}

function approveAllSuggestions() {
    saveState();
    const ss=document.querySelector('#resume-section-summary .suggested-item');
    if(ss){const b=ss.querySelector('button.bg-green-500');if(b)approveSummary(b,false);}
    document.querySelectorAll('#skills-container .suggested-item').forEach(s=>{const b=s.querySelector('button.bg-green-500');if(b)approveSkill(b,false);});
    document.querySelectorAll('#experience-container > div').forEach(block=>{const ul=block.querySelector('ul');if(!ul)return;const sb=ul.querySelectorAll('.suggested-item');if(sb.length>0){ul.querySelectorAll('li:not(.helper-bullet):not(.suggested-item)').forEach(ob=>ob.remove());sb.forEach(b=>{const btn=b.querySelector('button.bg-green-500');if(btn)approveBullet(btn,false);})}});
    checkPendingSuggestions(); showToast('All suggestions approved!');
}

function approveSummary(btn,doSave=true) {
    if(doSave)saveState();
    const div=btn.closest('.suggested-item');
    const text=div.querySelector('p').innerText;
    const op=div.closest('section').querySelector('p:not(.text-purple-900)');
    if(op)op.innerText=text; div.remove(); if(doSave)checkPendingSuggestions();
}

function approveSkill(btn,doSave=true) {
    if(doSave)saveState();
    const span=btn.closest('.suggested-item');
    const text=span.querySelector('span.outline-none').innerText;
    span.className='bg-gray-100 border border-gray-200 text-gray-800 px-3 py-1 rounded-md text-sm font-medium relative group flex items-center cursor-move';
    span.draggable=true; span.ondragstart=dragStart; span.ondragover=dragOver; span.ondragend=dragEnd;
    span.innerHTML=`<svg class="w-3 h-3 mr-1.5 text-gray-400 opacity-0 group-hover:opacity-100 pointer-events-none" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M4 8h16M4 16h16"></path></svg><span contenteditable="true" class="outline-none min-w-[20px] cursor-text" onmousedown="event.stopPropagation()">${text}</span><button type="button" contenteditable="false" onclick="removeElement(this.closest('.group'))" class="absolute -top-2 -right-2 bg-red-500 text-white rounded-full w-4 h-4 flex items-center justify-center text-xs opacity-0 group-hover:opacity-100 z-10 shadow no-print cursor-pointer">×</button>`;
    if(doSave)checkPendingSuggestions();
}

function approveBullet(btn,doSave=true) {
    if(doSave)saveState();
    const li=btn.closest('li');
    const text=li.querySelector('span.outline-none').innerText;
    li.className='relative group/bullet pr-6';
    li.innerHTML=`<span contenteditable="true" class="block outline-none">${text}</span><button type="button" contenteditable="false" onclick="removeElement(this.closest('li'))" class="absolute top-0.5 right-0 bg-red-400 text-white rounded-full w-4 h-4 flex items-center justify-center text-xs opacity-0 group-hover/bullet:opacity-100 z-10 shadow no-print cursor-pointer">×</button>`;
    if(doSave)checkPendingSuggestions();
}

async function generateCoverLetter() {
    const jobDesc=document.getElementById('cl-job-description-input').value.trim();
    const apiKey=document.getElementById('global-api-key-input').value.trim();
    const model=document.getElementById('global-ai-model-input').value.trim()||'gemini-2.5-flash';
    const errMsg=document.getElementById('cl-error-msg');
    if(!apiKey){errMsg.innerText='Set Gemini API Key first.';errMsg.classList.remove('hidden');return;}
    if(!jobDesc){errMsg.innerText='Paste a job description.';errMsg.classList.remove('hidden');return;}
    errMsg.classList.add('hidden');
    document.getElementById('cl-generate-btn').classList.add('hidden');
    document.getElementById('cl-job-description-input').disabled=true;
    document.getElementById('cl-loading').classList.remove('hidden');
    document.getElementById('cl-loading').classList.add('flex');
    document.getElementById('cl-output-container').classList.add('hidden');
    const resume=getScrapedResume();
    try {
        const url=`https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`;
        const result=await fetchWithRetry(url,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({contents:[{parts:[{text:`Job: "${jobDesc}"\nResume: ${JSON.stringify(resume,null,2)}\nWrite a professional cover letter in plain text.`}]}],systemInstruction:{parts:[{text:"You are an expert cover letter writer. Output only plain text."}]}})});
        const text=result.candidates?.[0]?.content?.parts?.[0]?.text||'';
        document.getElementById('cl-output-text').value=text;
        document.getElementById('cl-input-area').classList.add('hidden');
        document.getElementById('cl-output-container').classList.remove('hidden');
        document.getElementById('cl-output-container').classList.add('flex');
        document.getElementById('cl-download-group').classList.remove('hidden');
        document.getElementById('cl-download-group').classList.add('flex');
        showToast('Cover letter generated!');
    } catch(err) {
        document.getElementById('cl-job-description-input').disabled=false;
        errMsg.innerText='Error: '+err.message; errMsg.classList.remove('hidden');
    } finally {
        document.getElementById('cl-loading').classList.add('hidden');
        document.getElementById('cl-loading').classList.remove('flex');
        document.getElementById('cl-generate-btn').classList.remove('hidden');
        document.getElementById('cl-generate-btn').innerText='Regenerate';
        document.getElementById('cl-job-description-input').disabled=false;
    }
}

async function downloadCoverLetter() {
    const text=document.getElementById('cl-output-text').value;
    const format=document.getElementById('cl-format-select').value;
    let blob;
    if(format==='txt') blob=new Blob([text],{type:'text/plain'});
    else if(format==='html') blob=new Blob([`<!DOCTYPE html><html><head><meta charset="UTF-8"><title>Cover Letter</title><style>body{font-family:sans-serif;line-height:1.6;padding:2rem;max-width:800px;margin:0 auto;}</style></head><body>${text.replace(/\n/g,'<br>')}</body></html>`],{type:'text/html'});
    else if(format==='pdf'){try{const d=document.createElement('div');d.innerHTML=`<div style="font-family:sans-serif;line-height:1.6;padding:2rem;font-size:14px;">${text.replace(/\n/g,'<br>')}</div>`;blob=await html2pdf().set({margin:10,filename:'Cover_Letter.pdf',image:{type:'jpeg',quality:.8},html2canvas:{scale:2},jsPDF:{unit:'mm',format:'a4',orientation:'portrait'}}).from(d).output('blob');}catch(e){showToast('PDF error.',true);return;}}
    try {
        if(window.showSaveFilePicker){const mime={'txt':'text/plain','html':'text/html','pdf':'application/pdf'};const h=await window.showSaveFilePicker({suggestedName:`Cover_Letter.${format}`,types:[{description:`${format.toUpperCase()} File`,accept:{[mime[format]]:[`.${format}`]}}]});const w=await h.createWritable();await w.write(blob);await w.close();showToast('Saved!');}
        else{const url=URL.createObjectURL(blob);const a=Object.assign(document.createElement('a'),{href:url,download:`Cover_Letter.${format}`});document.body.appendChild(a);a.click();document.body.removeChild(a);URL.revokeObjectURL(url);showToast('Downloaded!');}
    } catch(e){if(e.name!=='AbortError')showToast('Save failed.',true);}
}
